# 1. Obtenemos el ID de cuenta dinámicamente
data "aws_caller_identity" "current" {}

# 2. EL TECHO (Permission Boundary)
# Esta es la política que LIMITARÁ a los usuarios creados por el Junior.
resource "aws_iam_policy" "boundary_policy" {
  name        = "Boundary-S3-EC2-Only"
  description = "Este es el techo maximo de permisos"
  policy      = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect   = "Allow"
      Action   = ["s3:*", "ec2:*"]
      Resource = "*"
    }]
  })
}

# 3. POLÍTICA DEL DELEGADO (Junior Admin)
# Esta política le da permisos de IAM pero lo obliga a usar el Boundary.
resource "aws_iam_policy" "delegated_policy" {
  name = "IAM-Junior-Admin-Policy"
  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      # Bloque A: Permitir crear usuarios y roles SOLO si incluyen el Boundary
      {
        Sid    = "AllowCreateWithBoundary"
        Effect = "Allow"
        Action = [
          "iam:CreateUser",
          "iam:CreateRole"
        ]
        Resource = "*"
        Condition = {
          "StringEquals" = {
            "iam:PermissionsBoundary" = aws_iam_policy.boundary_policy.arn
          }
        }
      },
      # Bloque B: Permitir poner el Boundary (Obligatorio para que funcione el A)
      {
        Sid    = "AllowPutBoundary"
        Effect = "Allow"
        Action = [
          "iam:PutUserPermissionsBoundary",
          "iam:PutRolePermissionsBoundary"
        ]
        Resource = "*"
        Condition = {
          "StringEquals" = {
            "iam:PermissionsBoundary" = aws_iam_policy.boundary_policy.arn
          }
        }
      },
      # Bloque C: Permisos generales para gestionar el usuario una vez creado
      {
        Sid    = "AllowGeneralIAMActions"
        Effect = "Allow"
        Action = [
          "iam:AttachUserPolicy",
          "iam:DetachUserPolicy",
          "iam:PutUserPolicy",
          "iam:Get*",
          "iam:List*"
        ]
        Resource = "*"
      }
    ]
  })
}

# 4. CREACIÓN DEL USUARIO DELEGADO Y ATTACHMENT
resource "aws_iam_user" "delegated_user" {
  name = "JuniorAdmin"
}

resource "aws_iam_user_policy_attachment" "attach_policy" {
  user       = aws_iam_user.delegated_user.name # Error corregido: nombre, no ARN
  policy_arn = aws_iam_policy.delegated_policy.arn
}
